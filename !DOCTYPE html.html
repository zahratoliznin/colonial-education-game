      startBGM() {
        if (!this.musicEnabled || this.isPlayingMusic) return;
        try {
          Tone.start();
          if (!this.synth) {
            this.synth = new Tone.PolySynth(Tone.Synth, {
              oscillator: { type: "sine" },
              envelope: { attack: 0.1, decay: 0.3, sustain: 0.6, release: 1.2 }
            }).toDestination();
            this.synth.volume.value = -16;

            this.bassSynth = new Tone.MonoSynth({
              oscillator: { type: "triangle" },
              envelope: { attack: 0.2, decay: 0.4, sustain: 0.8, release: 1.5 }
            }).toDestination();
            this.bassSynth.volume.value = -20;

            const chords = [
              { chord: ["C4", "E4", "G4", "B4"], bass: "C3" },
              { chord: ["A3", "C4", "E4", "G4"], bass: "A2" },
              { chord: ["F3", "A3", "C4", "E4"], bass: "F2" },
              { chord: ["G3", "B3", "D4", "F4"], bass: "G2" }
            ];

            let step = 0;
            this.bgMusicLoop = new Tone.Loop((time) => {
              const current = chords[step % chords.length];
              this.synth.triggerAttackRelease(current.chord, "2n", time);
              this.bassSynth.triggerAttackRelease(current.bass, "1n", time);
              step++;
            }, "1m").start(0);

            Tone.Transport.bpm.value = 82;
          }
          Tone.Transport.start();
          this.isPlayingMusic = true;
        } catch (e) {

          this.isPlayingMusic = false;

      playCatchSound() {
        if (!this.enabled) return;
        this.init();
        if (!this.ctx) return;

        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(880, this.ctx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.3, this.ctx.currentTime);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.15);

      playClappingSound() {
        if (!this.enabled) return;
        this.init();
        if (!this.ctx) return;

        const bufferSize = this.ctx.sampleRate * 2.5;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) {
          data[i] = Math.random() * 2 - 1;
        }

        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.value = 1200;
        filter.Q.value = 1.0;

        const gain = this.ctx.createGain();
        const now = this.ctx.currentTime;
        gain.gain.setValueAtTime(0.01, now);
        gain.gain.linearRampToValueAtTime(0.35, now + 0.3);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 2.5);

        noise.connect(filter);
        filter.connect(gain);
        gain.connect(this.ctx.destination);

        noise.start(now);
        noise.stop(now + 2.5);
      },

          osc.stop(this.ctx.currentTime + times[i] + durations[i]);
        });

    async function requestWebcamAccess() {
      const statusText = document.getElementById('tracking-status-text');
      const statusBox = document.getElementById('tracking-status');
      const statusDot = document.getElementById('tracking-dot');

      try {
        statusText.textContent = "Loading Model...";

        if (typeof Hands === 'undefined') {
          throw new Error('MediaPipe Hands script failed to load');
        }

        handsDetector = new Hands({
          locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`
        });

        handsDetector.setOptions({
          maxNumHands: 1,
          modelComplexity: 1,
          minDetectionConfidence: 0.2,
          minTrackingConfidence: 0.2
        });

        handsDetector.onResults(onHandsResults);

        statusText.textContent = "Cam Starting...";

        const stream = await navigator.mediaDevices.getUserMedia({
          video: {
            facingMode: 'user',
            width: { ideal: 640 },
            height: { ideal: 480 },
            frameRate: { ideal: 30 }
          },
          audio: false
        });

        pipVideo.srcObject = stream;
        await pipVideo.play();

        GAME_STATE.cameraActive = true;
        
        async function processFrame() {
          if (GAME_STATE.cameraActive && handsDetector && pipVideo.readyState >= 2) {
            try {
              await handsDetector.send({ image: pipVideo });
            } catch (err) {
              console.warn("Hand frame error:", err);
            }
          }
          if (GAME_STATE.cameraActive) {

    /* ==========================================================================
       GESTURE RECOGNITION & CURSOR
       ========================================================================== */
    function onHandsResults(results) {
      const statusText = document.getElementById('tracking-status-text');
      const statusDot = document.getElementById('tracking-dot');

      if (!results.multiHandLandmarks || results.multiHandLandmarks.length === 0) {
        if (GAME_STATE.trackingLossTimer > 0) {
          GAME_STATE.trackingLossTimer--;
        } else {
          if (GAME_STATE.trackingDetected) {
            GAME_STATE.trackingDetected = false;
            GAME_STATE.landmarks = null;
            if (statusText) statusText.textContent = "Ready - Show Hand";
            if (statusDot) statusDot.className = "w-2 h-2 rounded-full bg-emerald-400 animate-ping";
          }
        }
        return;
      }

      const handLandmarks = results.multiHandLandmarks[0];

      GAME_STATE.trackingLossTimer = 15;
      GAME_STATE.trackingDetected = true;
      GAME_STATE.landmarks = handLandmarks;

      if (statusText) statusText.textContent = "Hand Detected ✓";
      if (statusDot) statusDot.className = "w-2 h-2 rounded-full bg-emerald-400";

      const wrist = handLandmarks[0];
      const indexTip = handLandmarks[8];
      const indexPip = handLandmarks[6];
      const middleTip = handLandmarks[12];
      const middlePip = handLandmarks[10];
      const ringTip = handLandmarks[16];
      const ringPip = handLandmarks[14];
      const pinkyTip = handLandmarks[20];
      const pinkyPip = handLandmarks[18];

      const targetX = 1.0 - indexTip.x;
      const targetY = indexTip.y;

      const smoothing = 0.45;
      GAME_STATE.cursor.x += (targetX - GAME_STATE.cursor.x) * smoothing;
      GAME_STATE.cursor.y += (targetY - GAME_STATE.cursor.y) * smoothing;

      // 1. OPEN PALM (✋) - For catching falling objects
      const dIndexTip = Math.hypot(indexTip.x - wrist.x, indexTip.y - wrist.y);
      const dMiddleTip = Math.hypot(middleTip.x - wrist.x, middleTip.y - wrist.y);
      const dRingTip = Math.hypot(ringTip.x - wrist.x, ringTip.y - wrist.y);
      const dPinkyTip = Math.hypot(pinkyTip.x - wrist.x, pinkyTip.y - wrist.y);

      const dIndexPip = Math.hypot(indexPip.x - wrist.x, indexPip.y - wrist.y);
      const dMiddlePip = Math.hypot(middlePip.x - wrist.x, middlePip.y - wrist.y);
      const dRingPip = Math.hypot(ringPip.x - wrist.x, ringPip.y - wrist.y);
      const dPinkyPip = Math.hypot(pinkyPip.x - wrist.x, pinkyPip.y - wrist.y);

      const allExtended = (dIndexTip > dIndexPip * 1.1) &&
                          (dMiddleTip > dMiddlePip * 1.1) &&
                          (dRingTip > dRingPip * 1.1) &&
                          (dPinkyTip > dPinkyPip * 1.1);

      GAME_STATE.cursor.isOpenHand = allExtended;

      // 2. POINTING GESTURE (👉) - For selecting quiz options
      const indexExtended = dIndexTip > dIndexPip * 1.15;
      const othersCurled = (dMiddleTip < dMiddlePip * 1.05) &&
                           (dRingTip < dRingPip * 1.05) &&
                           (dPinkyTip < dPinkyPip * 1.05);

      GAME_STATE.cursor.isPointing = indexExtended && othersCurled && !allExtended;

      updateVirtualCursorPos();
    }

    function initFallbackMouseEvents() {
      window.addEventListener('mousemove', (e) => {
        if (!GAME_STATE.trackingDetected) {
          GAME_STATE.cursor.x = e.clientX / window.innerWidth;
          GAME_STATE.cursor.y = e.clientY / window.innerHeight;
          updateVirtualCursorPos();
        }
      });

      window.addEventListener('mousedown', () => {
        if (!GAME_STATE.trackingDetected) {
          if (GAME_STATE.mode === 'CATCHING') {
            GAME_STATE.cursor.isOpenHand = true;
          } else {
            GAME_STATE.cursor.isPointing = true;
          }
          triggerInteractionAtCursor();
        }
      });

      window.addEventListener('mouseup', () => {
        if (!GAME_STATE.trackingDetected) {
          GAME_STATE.cursor.isPointing = false;
          GAME_STATE.cursor.isOpenHand = false;
        }
      });

      window.addEventListener('touchstart', (e) => {
        if (!GAME_STATE.trackingDetected && e.touches.length > 0) {
          const touch = e.touches[0];
          GAME_STATE.cursor.x = touch.clientX / window.innerWidth;
          GAME_STATE.cursor.y = touch.clientY / window.innerHeight;
          if (GAME_STATE.mode === 'CATCHING') {
            GAME_STATE.cursor.isOpenHand = true;
          } else {
            GAME_STATE.cursor.isPointing = true;
          }
          updateVirtualCursorPos();
          triggerInteractionAtCursor();
        }
      }, { passive: true });

      window.addEventListener('touchmove', (e) => {
        if (!GAME_STATE.trackingDetected && e.touches.length > 0) {
          const touch = e.touches[0];
          GAME_STATE.cursor.x = touch.clientX / window.innerWidth;
          GAME_STATE.cursor.y = touch.clientY / window.innerHeight;
          updateVirtualCursorPos();
        }
      }, { passive: true });

      window.addEventListener('touchend', () => {
        if (!GAME_STATE.trackingDetected) {
          GAME_STATE.cursor.isPointing = false;
          GAME_STATE.cursor.isOpenHand = false;
        }
      });
    }

    function updateVirtualCursorPos() {
      const cursorEl = document.getElementById('dwell-cursor');
      const iconEl = document.getElementById('dwell-icon');
      if (!cursorEl) return;

      const pxX = GAME_STATE.cursor.x * window.innerWidth;
      const pxY = GAME_STATE.cursor.y * window.innerHeight;

      cursorEl.style.left = `${pxX}px`;
      cursorEl.style.top = `${pxY}px`;
      cursorEl.classList.remove('hidden');

      if (GAME_STATE.cursor.isOpenHand) {
        if (iconEl) iconEl.textContent = "✋";
        cursorEl.classList.add('scale-125', 'border-sky-400', 'bg-sky-500/30');
        cursorEl.classList.remove('border-amber-400', 'border-emerald-400', 'bg-emerald-500/30');
      } else if (GAME_STATE.cursor.isPointing) {
        if (iconEl) iconEl.textContent = "👉";
        cursorEl.classList.add('scale-125', 'border-emerald-400', 'bg-emerald-500/30');
        cursorEl.classList.remove('border-amber-400', 'border-sky-400', 'bg-sky-500/30');
      } else {
        if (iconEl) iconEl.textContent = "✋";
        cursorEl.classList.remove('scale-125', 'border-emerald-400', 'border-sky-400', 'bg-emerald-500/30', 'bg-sky-500/30');
        cursorEl.classList.add('border-amber-400');
      }

      checkInteractions(pxX, pxY);
    }

    function checkInteractions(pxX, pxY) {
      if (GAME_STATE.cooldown) return;

      if (GAME_STATE.mode === 'CATCHING') {
        // ONLY OPEN PALM can catch falling objects
        if (!GAME_STATE.cursor.isOpenHand && GAME_STATE.cameraActive) {
          return;
        }

        const artifacts = document.querySelectorAll('.floating-artifact');
        for (const artifact of artifacts) {
          const rect = artifact.getBoundingClientRect();
          const pad = 30;
          if (
            pxX >= rect.left - pad &&
            pxX <= rect.right + pad &&
            pxY >= rect.top - pad &&
            pxY <= rect.bottom + pad
          ) {
            catchArtifact(artifact);
            return;
          }
        }
        return;
      }

      if (GAME_STATE.mode === 'QUIZ') {
        // ONLY POINTING GESTURE can select answer options or next buttons when camera active
        if (GAME_STATE.cameraActive && !GAME_STATE.cursor.isPointing) {
          resetDwellProgress();
          return;
        }
      }

      if (GAME_STATE.cursor.isPointing || (!GAME_STATE.cameraActive)) {
        triggerInteractionAtCursor();
        return;
      }

      if (GAME_STATE.dwellEnabled && GAME_STATE.mode !== 'CATCHING') {
        const hoveredElem = document.elementFromPoint(pxX, pxY);
        const interactiveTarget = hoveredElem?.closest('[data-interactive="true"]');

        if (interactiveTarget) {
          if (GAME_STATE.dwellTarget !== interactiveTarget) {
            GAME_STATE.dwellTarget = interactiveTarget;
            GAME_STATE.dwellStartTime = Date.now();
          } else {
            const elapsed = Date.now() - GAME_STATE.dwellStartTime;
            const progress = Math.min(elapsed / GAME_STATE.dwellDuration, 1.0);
            const fill = document.getElementById('dwell-fill');
            if (fill) fill.style.transform = `scale(${progress})`;

            if (progress >= 1.0) {
              triggerInteractiveElement(interactiveTarget);
              resetDwellProgress();
            }
          }
        } else {
          resetDwellProgress();
        }
      } else {
        resetDwellProgress();
      }
    }

    function resetDwellProgress() {
      GAME_STATE.dwellTarget = null;
      GAME_STATE.dwellStartTime = 0;
      const fill = document.getElementById('dwell-fill');
      if (fill) fill.style.transform = 'scale(0)';
    }

    function triggerInteractionAtCursor() {
      if (GAME_STATE.cooldown) return;
      const pxX = GAME_STATE.cursor.x * window.innerWidth;
      const pxY = GAME_STATE.cursor.y * window.innerHeight;
      const hoveredElem = document.elementFromPoint(pxX, pxY);
      const interactiveTarget = hoveredElem?.closest('[data-interactive="true"]');

      if (interactiveTarget) triggerInteractiveElement(interactiveTarget);
    }

    function triggerInteractiveElement(elem) {
      const action = elem.getAttribute('data-action');
      const param = elem.getAttribute('data-param');
      if (!action) return;

      GAME_STATE.cooldown = true;
      setTimeout(() => GAME_STATE.cooldown = false, 500);

      if (action === 'start-game') {
        startGame();
      } else if (action === 'catch-artifact') {
        catchArtifact(elem);
      } else if (action === 'select-quiz') {
        submitQuizAnswer(parseInt(param, 10));
      } else if (action === 'next-question') {
        resumeCatchingOrFinish();
      } else if (action === 'restart-game') {
        resetGame();
      }
    }

    /* ==========================================================================
       CATCHING GAME ENGINE
       ========================================================================== */
    function startGame() {
      AudioEngine.init();
      AudioEngine.startBGM();
      GAME_STATE.score = 0;
      GAME_STATE.answeredCount = 0;
      GAME_STATE.mode = 'CATCHING';
      prepareQuestions();
      renderUI();
      spawnArtifacts();
    }

    function spawnArtifacts() {
      const container = document.getElementById('floating-artifacts-layer');
      if (!container) return;
      container.innerHTML = '';

      if (GAME_STATE.mode !== 'CATCHING') return;

      const numArtifacts = 4;
      const placedPositions = [];

      for (let i = 0; i < numArtifacts; i++) {
        let leftPercent = 10;
        let attempts = 0;
        let valid = false;

        while (!valid && attempts < 25) {
          leftPercent = 10 + Math.random() * 75;
          valid = placedPositions.every(pos => Math.abs(pos - leftPercent) >= 18);
          attempts++;
        }

        placedPositions.push(leftPercent);
        createArtifactElement(container, leftPercent, i * 1.5);
      }
    }

    function createArtifactElement(container, leftPercent, delaySeconds = 0) {
      const artifact = document.createElement('div');
      const icon = ARTIFACT_ICONS[Math.floor(Math.random() * ARTIFACT_ICONS.length)];
      const duration = 6.5 + Math.random() * 3.5;

      artifact.className = "floating-artifact pointer-events-auto p-4 rounded-2xl bg-stone-900/90 border-2 border-sepia-500 shadow-2xl flex items-center justify-center text-4xl hover:scale-125 transition transform";
      artifact.style.left = `${leftPercent}%`;
      artifact.style.animationDuration = `${duration}s`;
      artifact.style.animationDelay = `${delaySeconds}s`;
      artifact.setAttribute('data-interactive', 'true');
      artifact.setAttribute('data-action', 'catch-artifact');
      artifact.innerHTML = `<span>${icon}</span>`;

      container.appendChild(artifact);
    }

    function catchArtifact(elem) {
      if (GAME_STATE.mode !== 'CATCHING') return;

      GAME_STATE.cooldown = true;
      setTimeout(() => GAME_STATE.cooldown = false, 600);

      AudioEngine.playCatchSound();

      elem.style.transform = "scale(1.8) rotate(20deg)";
      elem.style.opacity = "0";

      setTimeout(() => {
        if (elem.parentNode) elem.parentNode.removeChild(elem);
      }, 200);

      triggerRandomQuestion();
    }

    function triggerRandomQuestion() {
      if (GAME_STATE.questionQueue.length === 0) prepareQuestions();

      GAME_STATE.activeQuestion = GAME_STATE.questionQueue.pop();
      GAME_STATE.activeQuestion.userAnswer = null;
      GAME_STATE.mode = 'QUIZ';

      const container = document.getElementById('floating-artifacts-layer');
      if (container) container.innerHTML = '';

      renderUI();
    }

    function submitQuizAnswer(ansIdx) {
      if (GAME_STATE.activeQuestion.userAnswer !== null) return;

      GAME_STATE.activeQuestion.userAnswer = ansIdx;
      const isCorrect = (ansIdx === GAME_STATE.activeQuestion.answer);

      if (isCorrect) {
        GAME_STATE.score += 10;
        AudioEngine.playCorrectSound();
      } else {
        AudioEngine.playWrongSound();
      }

      GAME_STATE.answeredCount++;
      renderUI();
    }

    function resumeCatchingOrFinish() {
      if (GAME_STATE.answeredCount >= 10) {
        GAME_STATE.mode = 'RESULTS';
        renderUI();
        triggerVictoryEffects();
      } else {
        GAME_STATE.mode = 'CATCHING';
        renderUI();
        spawnArtifacts();
      }
    }

    function resetGame() {
      AudioEngine.stopContinuousVictoryLoop();
      GAME_STATE.score = 0;
      GAME_STATE.answeredCount = 0;
      GAME_STATE.mode = 'START';
      renderUI();
    }

    function triggerVictoryEffects() {
      AudioEngine.startContinuousVictoryLoop();

      if (typeof confetti === 'function') {
        window.isVictoryActive = true;

        (function continuousConfettiFrame() {
          if (!window.isVictoryActive || GAME_STATE.mode !== 'RESULTS') return;

          confetti({
            particleCount: 8,
            angle: 60,
            spread: 55,
            origin: { x: 0 },

          confetti({
            particleCount: 8,
            angle: 120,
            spread: 55,
            origin: { x: 1 },
            colors: ['#f59e0b', '#10b981', '#3b82f6', '#d97706']
          });

          setTimeout(continuousConfettiFrame, 350);
        })();
      }
    }

       ANIMATION LOOP & LANDMARK OVERLAY
       ========================================================================== */
    function gameLoop() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      if (GAME_STATE.trackingDetected && GAME_STATE.landmarks) {
        drawHandOverlayOnMainCanvas(GAME_STATE.landmarks);
      }

      requestAnimationFrame(gameLoop);
    }

    function drawHandOverlayOnMainCanvas(landmarks) {
      if (!landmarks) return;

      const HAND_CONNECTIONS = [
        [0,1],[1,2],[2,3],[3,4],
        [0,5],[5,6],[6,7],[7,8],
        [5,9],[9,10],[10,11],[11,12],
        [9,13],[13,14],[14,15],[15,16],
        [13,17],[17,18],[18,19],[19,20],
        [0,17]
      ];

      ctx.lineWidth = 4;
      if (GAME_STATE.cursor.isOpenHand) {
        ctx.strokeStyle = 'rgba(56, 189, 248, 0.9)';
      } else if (GAME_STATE.cursor.isPointing) {
        ctx.strokeStyle = 'rgba(52, 211, 153, 0.9)';
      } else {
        ctx.strokeStyle = 'rgba(245, 158, 11, 0.8)';
      }

      for (const [start, end] of HAND_CONNECTIONS) {
        const p1 = landmarks[start];
        const p2 = landmarks[end];

        const x1 = (1.0 - p1.x) * canvas.width;
        const y1 = p1.y * canvas.height;
        const x2 = (1.0 - p2.x) * canvas.width;
        const y2 = p2.y * canvas.height;

        ctx.beginPath();
        ctx.moveTo(x1, y1);
        ctx.lineTo(x2, y2);
        ctx.stroke();
      }

      for (let i = 0; i < landmarks.length; i++) {
        const lm = landmarks[i];
        const x = (1.0 - lm.x) * canvas.width;
        const y = lm.y * canvas.height;

        ctx.beginPath();
        if (i === 4 || i === 8 || i === 12 || i === 16 || i === 20) {
          ctx.arc(x, y, 8, 0, Math.PI * 2);
          if (GAME_STATE.cursor.isOpenHand) {
            ctx.fillStyle = '#38bdf8';
          } else if (GAME_STATE.cursor.isPointing) {
            ctx.fillStyle = '#34d399';
          } else {
            ctx.fillStyle = '#f59e0b';
          }
        } else {
          ctx.arc(x, y, 5, 0, Math.PI * 2);
          ctx.fillStyle = 'rgba(251, 191, 36, 0.9)';
        }
        ctx.fill();
      }
    }

        AudioEngine.stopBGM();
      }

          setTimeout(() => btn.textContent = "Copy Link", 2000);
        }
